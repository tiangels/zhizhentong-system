### 向量化服务的设计原理

#### 1. **容错机制设计**

```python
class VectorService:
    def __init__(self):
        # 优先使用构建知识库模块
        self.use_knowledge_base_service = True

        # 备用本地模型
        self.local_model = None

        # 最后备用方案
        self.tfidf_model = None

    def text_to_vector(self, text: str) -> np.ndarray:
        try:
            # 方案1: 调用embedding_models服务
            if self.use_knowledge_base_service:
                return self._call_knowledge_base_service(text)
        except:
            pass

        try:
            # 方案2: 使用本地模型
            if self.local_model:
                return self._use_local_model(text)
        except:
            pass

        try:
            # 方案3: 使用TF-IDF
            if self.tfidf_model:
                return self._use_tfidf(text)
        except:
            pass

        # 方案4: 返回零向量
        return np.zeros(768)
```

## 🌊 流式处理技术

### 功能概述

RAG 系统新增了两个重要功能：

1. **内容摘要功能** - 将检索到的内容先摘要到 50 字以内，提高回答质量
2. **流式输出功能** - 支持实时流式响应，提供更好的用户体验

### 技术实现

#### 1. 内容摘要功能

```python
def _summarize_content(self, content: str, max_length: int = 50) -> str:
    """将内容摘要到指定长度"""
    try:
        # 使用T5模型进行摘要
        summary = self.summarization_service.summarize_medical_text(
            text=content,
            max_length=max_length
        )
        return summary
    except Exception as e:
        logger.warning(f"摘要失败，返回截断原文: {e}")
        # 摘要失败时返回截断原文
        return content[:max_length] + "..." if len(content) > max_length else content
```

#### 2. 流式输出功能

```python
async def query_stream(self, question: str, top_k: int = 5,
                      response_type: str = "general",
                      similarity_threshold: float = 200.0):
    """流式查询方法"""
    try:
        # 状态更新
        yield {"type": "status", "message": "开始处理您的问题..."}

        # 用户数据处理
        yield {"type": "status", "message": "正在分析问题..."}
        processed_question = self._preprocess_question(question)

        # 向量化
        yield {"type": "status", "message": "正在向量化处理..."}
        query_vector = self.vector_service.text_to_vector(processed_question)

        # 检索
        yield {"type": "status", "message": "正在检索相关知识..."}
        retrieved_docs = self.retrieval_service.retrieve_documents(
            query_vector, top_k=top_k
        )

        # 内容摘要
        yield {"type": "status", "message": "正在整理检索到的内容..."}
        summarized_context = self._summarize_content(
            self._build_context(retrieved_docs)
        )

        # 流式生成
        yield {"type": "status", "message": "正在生成回答..."}
        async for chunk in self.llm_service.generate_stream(
            prompt=self._build_prompt(question, summarized_context, response_type)
        ):
            yield {"type": "content", "content": chunk}

        # 文档展示
        yield {"type": "documents", "documents": retrieved_docs}

        # 完成通知
        yield {"type": "complete", "message": "回答生成完成"}

    except Exception as e:
        logger.error(f"流式查询失败: {e}")
        yield {"type": "error", "message": f"查询失败: {str(e)}"}
```

#### 3. LLM 流式生成

```python
async def generate_stream(self, prompt: str, max_tokens: int = 1000) -> AsyncGenerator[str, None]:
    """流式生成回答"""
    try:
        # 使用异步生成器
        async for token in self.model.generate_async(prompt, max_tokens=max_tokens):
            yield token
    except Exception as e:
        logger.error(f"流式生成失败: {e}")
        yield f"生成失败: {str(e)}"
```

```python
@app.post("/query/stream")

    return StreamingResponse(
        generate_response(),
        media_type="text/plain",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "Content-Type": "text/event-stream"
        }
    )
```

## 🗄️ ChromaDB 迁移

### 迁移概述

将 knowledge_retrieval_service 中的 FAISS 向量数据库替换为 ChromaDB，以统一整个系统的向量数据库使用。

### 主要更改

#### 1. vector_service.py

```python
class VectorService:
    def __init__(self):
        # 旧版本: FAISS索引
        # self.faiss_index = faiss.IndexFlatIP(vector_dim)

        # 新版本: ChromaDB实例
        self.chroma_client = chromadb.PersistentClient(path="./chroma_db")
        self.collection = self.chroma_client.get_or_create_collection(
            name="medical_knowledge"
        )

    def add_vectors_to_db(self, vectors: np.ndarray, metadata: List[Dict]):
        # 旧版本: FAISS添加
        # self.faiss_index.add(vectors.astype('float32'))

        # 新版本: ChromaDB添加
        self.collection.add(
            embeddings=vectors.tolist(),
            metadatas=metadata,
            ids=[f"doc_{i}" for i in range(len(vectors))]
        )

    def search_similar_vectors(self, query_vector: np.ndarray, top_k: int = 10):
        # 旧版本: FAISS搜索
        # scores, indices = self.faiss_index.search(query_vector.reshape(1, -1), top_k)

        # 新版本: ChromaDB搜索
        results = self.collection.query(
            query_embeddings=[query_vector.tolist()],
            n_results=top_k
        )
        return results['documents'][0], results['distances'][0]
```

#### 2. retrieval_service.py

```python
class RetrievalService:
    def __init__(self):
        # 集成ChromaDB和HuggingFaceEmbeddings
        self.chroma_client = chromadb.PersistentClient(path="./chroma_db")
        self.collection = self.chroma_client.get_or_create_collection(
            name="medical_knowledge"
        )
        self.embeddings = HuggingFaceEmbeddings(
            model_name="sentence-transformers/all-MiniLM-L6-v2"
        )

    def add_documents(self, vectors: np.ndarray, documents: List[Dict]):
        # 同时支持ChromaDB和本地numpy数组存储
        self.collection.add(
            embeddings=vectors.tolist(),
            documents=[doc['content'] for doc in documents],
            metadatas=[doc.get('metadata', {}) for doc in documents],
            ids=[doc.get('id', f"doc_{i}") for i, doc in enumerate(documents)]
        )

        # 备用存储
        self.backup_vectors = np.vstack([self.backup_vectors, vectors])
        self.backup_documents.extend(documents)

    def semantic_retrieval(self, query_vector: np.ndarray, top_k: int):
        # 使用ChromaDB的相似度搜索API
        results = self.collection.query(
            query_embeddings=[query_vector.tolist()],
            n_results=top_k
        )

        # 统一返回格式
        retrieved_docs = []
        for i, (doc, metadata, distance) in enumerate(zip(
            results['documents'][0],
            results['metadatas'][0],
            results['distances'][0]
        )):
            retrieved_docs.append({
                'content': doc,
                'metadata': metadata,
                'score': 1 - distance,  # 转换为相似度分数
                'rank': i + 1
            })

        return retrieved_docs
```

### 推荐方案：后端 API 层存储

**位置**: `codes/backend/app/modules/conversation/`

**优势**:

1. **职责清晰**: 后端负责业务逻辑和数据持久化
2. **数据一致性**: 统一的数据库管理，避免数据分散
3. **安全性**: 后端可以控制数据访问权限
4. **可扩展性**: 支持多用户、多会话管理
5. **审计追踪**: 完整的操作日志和用户行为记录

### 不推荐：RAG 服务层存储

**原因**:

1. **职责混乱**: RAG 服务专注于检索和生成，不应承担数据存储职责
2. **数据分散**: 对话数据与业务数据分离，难以统一管理
3. **扩展困难**: 难以支持多用户、权限控制等业务需求
4. **维护复杂**: 数据备份、恢复、迁移等操作复杂

### 推荐架构设计

```python
# codes/backend/app/modules/conversation/
class ConversationService:
    def __init__(self):
        """初始化对话服务"""
        self.db = get_database()
        self.rag_client = RAGClient("http://localhost:8002")

    def save_conversation(self, user_id: str, messages: List[Dict]) -> str:
        """保存对话记录"""
        conversation = {
            "user_id": user_id,
            "messages": messages,
            "created_at": datetime.now(),
            "updated_at": datetime.now()
        }

        # 保存到数据库
        conversation_id = self.db.conversations.insert_one(conversation).inserted_id

        # 调用RAG服务生成回答
        response = self.rag_client.chat(messages)

        # 更新对话记录
        self.db.conversations.update_one(
            {"_id": conversation_id},
            {"$push": {"messages": response}, "$set": {"updated_at": datetime.now()}}
        )

        return str(conversation_id)

    def get_conversation_history(self, user_id: str, limit: int = 10) -> List[Dict]:
        """获取对话历史"""
        conversations = self.db.conversations.find(
            {"user_id": user_id}
        ).sort("updated_at", -1).limit(limit)

        return list(conversations)

    def delete_conversation(self, conversation_id: str) -> bool:
        """删除对话记录"""
        result = self.db.conversations.delete_one({"_id": ObjectId(conversation_id)})
        return result.deleted_count > 0
```
