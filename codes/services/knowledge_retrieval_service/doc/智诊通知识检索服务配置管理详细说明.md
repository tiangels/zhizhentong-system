# 智诊通知识检索服务配置管理详细说明

## 📋 配置管理概述

配置管理是智诊通知识检索服务的核心组件，负责统一管理所有服务的配置信息，包括模型配置、服务配置、数据库配置等。该模块提供了灵活的配置管理机制，支持配置验证、动态更新和路径管理。

### 🎯 核心功能

- **统一配置管理**：集中管理所有服务配置
- **配置验证**：验证配置的有效性和完整性
- **动态更新**：支持运行时配置更新
- **路径管理**：统一管理模型路径和数据路径
- **环境适配**：支持不同环境的配置
- **安全配置**：敏感信息的安全管理
- **配置备份**：配置变更的备份和恢复

## 🔧 配置详解

### 1. 向量服务配置 (vector_service)

#### 配置说明

```json
{
  "vector_service": {
    "device": "cpu",
    "vector_dim": 512,
    "batch_size": 32,
    "text_model_path": "sentence-transformers/all-MiniLM-L6-v2",
    "image_model_path": "clip-ViT-B-32",
    "use_knowledge_base_service": true
  }
}
```

#### 参数详解

| 参数                         | 类型    | 默认值 | 说明                    |
| ---------------------------- | ------- | ------ | ----------------------- |
| `device`                     | string  | "cpu"  | 计算设备，支持 cpu/cuda |
| `vector_dim`                 | int     | 512    | 向量维度                |
| `batch_size`                 | int     | 32     | 批处理大小              |
| `text_model_path`            | string  | -      | 文本向量化模型路径      |
| `image_model_path`           | string  | -      | 图像向量化模型路径      |
| `use_knowledge_base_service` | boolean | true   | 是否使用知识库服务      |

#### 配置示例

```json
{
  "vector_service": {
    "device": "cuda",
    "vector_dim": 768,
    "batch_size": 64,
    "text_model_path": "text2vec-base-chinese",
    "image_model_path": "clip-vit-base-patch32",
    "use_knowledge_base_service": true
  }
}
```

```python
class ConfigManager:
    def __init__(self, config_path: str = None):
        """初始化配置管理器"""
        self.config_path = config_path or "api/config/retrieval_config.json"
        self.config = self.load_config()

    def load_config(self) -> Dict[str, Any]:
        """加载配置文件"""
        try:
            with open(self.config_path, 'r', encoding='utf-8') as f:
                return json.load(f)
        except Exception as e:
            logger.error(f"加载配置文件失败: {e}")
            return self.get_default_config()

    def save_config(self) -> bool:
        """保存配置文件"""
        try:
            with open(self.config_path, 'w', encoding='utf-8') as f:
                json.dump(self.config, f, indent=2, ensure_ascii=False)
            return True
        except Exception as e:
            logger.error(f"保存配置文件失败: {e}")
            return False

    def get_vector_service_config(self) -> Dict[str, Any]:
        """获取向量服务配置"""
        return self.config.get("vector_service", {})

    def get_retrieval_service_config(self) -> Dict[str, Any]:
        """获取检索服务配置"""
        return self.config.get("retrieval_service", {})

    def get_llm_service_config(self) -> Dict[str, Any]:
        """获取LLM服务配置"""
        return self.config.get("llm_service", {})

    def get_summarization_service_config(self) -> Dict[str, Any]:
        """获取摘要服务配置"""
        return self.config.get("summarization_service", {})

    def get_api_config(self) -> Dict[str, Any]:
        """获取API服务配置"""
        return self.config.get("api", {})

    def update_config(self, section: str, key: str, value: Any) -> bool:
        """更新配置"""
        try:
            if section not in self.config:
                self.config[section] = {}
            self.config[section][key] = value
            return self.save_config()
        except Exception as e:
            logger.error(f"更新配置失败: {e}")
            return False

    def validate_config(self) -> bool:
        """验证配置"""
        try:
            # 验证必要配置
            required_sections = ["vector_service", "retrieval_service", "llm_service", "api"]
            for section in required_sections:
                if section not in self.config:
                    logger.error(f"缺少必要配置节: {section}")
                    return False

            # 验证向量服务配置
            vector_config = self.config["vector_service"]
            if "vector_dim" not in vector_config:
                logger.error("缺少向量维度配置")
                return False

            # 验证检索服务配置
            retrieval_config = self.config["retrieval_service"]
            if "max_results" not in retrieval_config:
                logger.error("缺少最大结果数配置")
                return False

            return True
        except Exception as e:
            logger.error(f"配置验证失败: {e}")
            return False

    def get_default_config(self) -> Dict[str, Any]:
        """获取默认配置"""
        return {
            "vector_service": {
                "device": "cpu",
                "vector_dim": 512,
                "batch_size": 32,
                "text_model_path": "sentence-transformers/all-MiniLM-L6-v2",
                "image_model_path": "clip-ViT-B-32",
                "use_knowledge_base_service": True
            },
            "retrieval_service": {
                "vector_dim": 512,
                "max_results": 20,
                "similarity_threshold": 0.5,
                "retrieval_strategy": "hybrid",
```

## 🔒 安全配置

### 1. 敏感信息管理

```python
import os
from cryptography.fernet import Fernet

class SecureConfigManager:
    def __init__(self, config_path: str, key_path: str = None):
        """初始化安全配置管理器"""
        self.config_path = config_path
        self.key_path = key_path or "config/encryption.key"
        self.cipher = self._get_cipher()

    def _get_cipher(self) -> Fernet:
        """获取加密器"""
        if os.path.exists(self.key_path):
            with open(self.key_path, 'rb') as f:
                key = f.read()
        else:
            key = Fernet.generate_key()
            with open(self.key_path, 'wb') as f:
                f.write(key)
        return Fernet(key)

    def encrypt_config(self, config: Dict[str, Any]) -> bytes:
        """加密配置"""
        config_str = json.dumps(config)
        return self.cipher.encrypt(config_str.encode())

    def decrypt_config(self, encrypted_config: bytes) -> Dict[str, Any]:
        """解密配置"""
        decrypted_str = self.cipher.decrypt(encrypted_config).decode()
        return json.loads(decrypted_str)
```

```python
class VersionedConfigManager:
    def __init__(self, config_path: str, backup_dir: str = "config/backups"):

    def save_config_version(self) -> str:
        """保存配置版本"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        version_file = os.path.join(self.backup_dir, f"config_{timestamp}.json")

        try:
            with open(self.config_path, 'r', encoding='utf-8') as f:
                config_content = f.read()

            with open(version_file, 'w', encoding='utf-8') as f:

            config_hash = hashlib.md5(config_content.encode()).hexdigest()

            # 保存版本信息
            version_info = {
                "timestamp": timestamp,
                "hash": config_hash,
                "file": version_file
            }

            version_info_file = os.path.join(self.backup_dir, f"version_{timestamp}.json")

            return timestamp

        except Exception as e:

            return True
```
