## 1. 模块概述

### 1.1 功能定位

分析器模块是智诊通向量化服务的核心组件，负责对预处理后的数据进行深度分析和特征提取。

### 1.2 架构设计

- **分层架构**: 基础分析器 → 专业分析器 → 综合分析器
- **统一接口**: 所有分析器都继承自 BaseAnalyzer
- **标准化输出**: 统一的分析结果格式

### 1.3 主要特性

- **多模态分析**: 支持文本、图像、语音等多种数据类型
- **医学专业**: 针对医疗场景优化的分析算法
- **高性能**: 支持并行处理和批量分析
- **可扩展**: 模块化设计，易于扩展新的分析器

## 2. 核心分析器

### 2.1 BaseAnalyzer 基础分析器

#### 2.1.1 文件概述

- **文件路径**: `analyzers/base_analyzer.py`
- **主要功能**: 提供所有分析器的统一接口和基础功能
- **设计模式**: 抽象基类模式

#### 2.1.2 核心接口

```python
class BaseAnalyzer(ABC):
    """基础分析器抽象类"""

    def __init__(self, config: Dict[str, Any]):
        """
        初始化分析器

        Args:
            config: 分析器配置参数
        """
        self.config = config
        self.logger = logging.getLogger(self.__class__.__name__)
        self.model = None
        self._load_model()

    @abstractmethod
    def analyze(self, data: Any) -> AnalysisResult:
        """
        分析数据

        Args:
            data: 待分析的数据

        Returns:
            AnalysisResult: 分析结果
        """
        pass

    @abstractmethod
    def _load_model(self):
        """加载分析模型"""
        pass

    def batch_analyze(self, data_list: List[Any]) -> List[AnalysisResult]:
        """
        批量分析数据

        Args:
            data_list: 待分析的数据列表

        Returns:
            List[AnalysisResult]: 分析结果列表
        """
        results = []
        for data in data_list:
            try:
                result = self.analyze(data)
                results.append(result)
            except Exception as e:
                self.logger.error(f"分析失败: {e}")
                results.append(AnalysisResult.error(str(e)))
        return results
```

#### 2.1.3 分析结果结构

```python
@dataclass
class AnalysisResult:
    """分析结果数据结构"""

    # 基本信息
    success: bool
    data_type: str
    timestamp: datetime

    # 分析内容
    features: Dict[str, Any]
    confidence: float
    metadata: Dict[str, Any]

    # 错误信息
    error_message: Optional[str] = None

    @classmethod
    def success(cls, data_type: str, features: Dict[str, Any],
                confidence: float, metadata: Dict[str, Any] = None):
        """创建成功结果"""
        return cls(
            success=True,
            data_type=data_type,
            timestamp=datetime.now(),
            features=features,
            confidence=confidence,
            metadata=metadata or {}
        )

    @classmethod
    def error(cls, error_message: str):
        """创建错误结果"""
        return cls(
            success=False,
            data_type="unknown",
            timestamp=datetime.now(),
            features={},
            confidence=0.0,
            metadata={},
            error_message=error_message
        )
```

```python
class TextAnalyzer(BaseAnalyzer):
    """文本分析器"""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.medical_terms = self._load_medical_terms()
        self.sentiment_analyzer = self._init_sentiment_analyzer()
        self.entity_extractor = self._init_entity_extractor()

    def analyze(self, text_data: str) -> AnalysisResult:
        """
        分析文本数据

        Args:
            text_data: 待分析的文本数据

        Returns:
            AnalysisResult: 文本分析结果
        """
        try:
            # 基础文本分析
            basic_features = self._analyze_basic_features(text_data)

            # 医学实体提取
            medical_entities = self._extract_medical_entities(text_data)

            # 情感分析
            sentiment = self._analyze_sentiment(text_data)

            # 关键词提取
            keywords = self._extract_keywords(text_data)

            # 文本质量评估
            quality_score = self._assess_text_quality(text_data)

            # 合并特征
            features = {
                'basic_features': basic_features,
                'medical_entities': medical_entities,
                'sentiment': sentiment,
                'keywords': keywords,
                'quality_score': quality_score
            }

            # 计算置信度
            confidence = self._calculate_confidence(features)

            return AnalysisResult.success(
                data_type='text',
                features=features,
                confidence=confidence,
                metadata={'text_length': len(text_data)}
            )

        except Exception as e:
            self.logger.error(f"文本分析失败: {e}")
            return AnalysisResult.error(str(e))
```

#### 3.1.3 医学实体提取

```python
def _extract_medical_entities(self, text: str) -> Dict[str, List[str]]:
    """
    提取医学实体

    Args:
        text: 输入文本

    Returns:
        Dict[str, List[str]]: 医学实体字典
    """
    entities = {
        'diseases': [],      # 疾病名称
        'symptoms': [],      # 症状
        'medications': [],   # 药物
        'procedures': [],    # 医疗程序
        'body_parts': [],    # 身体部位
        'test_results': []   # 检验结果
    }

    # 使用NER模型提取实体
    ner_results = self.entity_extractor.extract(text)

    for entity in ner_results:
        entity_type = entity.get('type', '').lower()
        entity_text = entity.get('text', '')

        if entity_type in entities:
            entities[entity_type].append(entity_text)

    return entities
```

#### 3.1.4 文本质量评估

```python
def _assess_text_quality(self, text: str) -> Dict[str, float]:
    """
    评估文本质量

    Args:
        text: 输入文本

    Returns:
        Dict[str, float]: 质量评估结果
    """
    quality_metrics = {
        'readability': 0.0,      # 可读性
        'completeness': 0.0,     # 完整性
        'medical_relevance': 0.0, # 医学相关性
        'coherence': 0.0,        # 连贯性
        'overall_score': 0.0     # 总体评分
    }

    # 可读性评估
    quality_metrics['readability'] = self._calculate_readability(text)

    # 完整性评估
    quality_metrics['completeness'] = self._assess_completeness(text)

    # 医学相关性评估
    quality_metrics['medical_relevance'] = self._assess_medical_relevance(text)

    # 连贯性评估
    quality_metrics['coherence'] = self._assess_coherence(text)

    # 计算总体评分
    quality_metrics['overall_score'] = np.mean([
        quality_metrics['readability'],
        quality_metrics['completeness'],
        quality_metrics['medical_relevance'],
        quality_metrics['coherence']
    ])

    return quality_metrics
```

#### 4.1.3 医学目标检测

```python
def _detect_medical_objects(self, image: np.ndarray) -> Dict[str, List[Dict]]:
    """
    检测医学目标

    Args:
        image: 输入图像

    Returns:
        Dict[str, List[Dict]]: 检测到的医学目标
    """
    detections = {
        'organs': [],           # 器官
        'lesions': [],          # 病变
        'medical_devices': [],  # 医疗设备
        'anatomical_landmarks': [] # 解剖标志
    }

    # 使用医学目标检测模型
    detection_results = self.medical_detector.detect(image)

    for detection in detection_results:
        obj_type = detection.get('class', '').lower()
        bbox = detection.get('bbox', [])
        confidence = detection.get('confidence', 0.0)

        detection_info = {
            'bbox': bbox,
            'confidence': confidence,
            'area': (bbox[2] - bbox[0]) * (bbox[3] - bbox[1])
        }

        if obj_type in detections:
            detections[obj_type].append(detection_info)

    return detections
```

```python
def _assess_image_quality(self, image: np.ndarray) -> Dict[str, float]:
    """
    评估图像质量

    Args:
        image: 输入图像

    Returns:
        Dict[str, float]: 质量评估结果
    """
    quality_metrics = {
        'sharpness': 0.0,        # 清晰度
        'contrast': 0.0,         # 对比度
        'brightness': 0.0,       # 亮度
        'noise_level': 0.0,      # 噪声水平
        'artifacts': 0.0,        # 伪影
        'overall_quality': 0.0   # 总体质量
    }

    # 清晰度评估
    quality_metrics['sharpness'] = self._calculate_sharpness(image)

    # 对比度评估
    quality_metrics['contrast'] = self._calculate_contrast(image)

    # 亮度评估
    quality_metrics['brightness'] = self._calculate_brightness(image)

    # 噪声水平评估
    quality_metrics['noise_level'] = self._calculate_noise_level(image)

    # 伪影检测
    quality_metrics['artifacts'] = self._detect_artifacts(image)

    # 计算总体质量
    quality_metrics['overall_quality'] = np.mean([
        quality_metrics['sharpness'],
        quality_metrics['contrast'],
        1.0 - quality_metrics['noise_level'],
        1.0 - quality_metrics['artifacts']
    ])

    return quality_metrics
```

#### 5.1.4 情感分析

```python
def _analyze_emotion(self, audio: np.ndarray) -> Dict[str, float]:
    """
    分析语音情感

    Args:
        audio: 音频数据

    Returns:
        Dict[str, float]: 情感分析结果
    """
    emotions = {
        'happy': 0.0,      # 高兴
        'sad': 0.0,        # 悲伤
        'angry': 0.0,      # 愤怒
        'fear': 0.0,       # 恐惧
        'surprise': 0.0,   # 惊讶
        'disgust': 0.0,    # 厌恶
        'neutral': 0.0,    # 中性
        'dominant_emotion': 'neutral'  # 主导情感
    }

    try:
        # 使用情感分析模型
        emotion_result = self.emotion_analyzer.analyze(audio)

        for emotion, score in emotion_result.items():
            if emotion in emotions:
                emotions[emotion] = score

        # 确定主导情感
        emotions['dominant_emotion'] = max(emotions, key=lambda x: emotions[x] if x != 'dominant_emotion' else 0)

    except Exception as e:
        self.logger.error(f"情感分析失败: {e}")

    return emotions
```

#### 5.1.5 声学特征提取

```python
def _extract_acoustic_features(self, audio: np.ndarray) -> Dict[str, Any]:
    """
    提取声学特征

    Args:
        audio: 音频数据

    Returns:
        Dict[str, Any]: 声学特征
    """
    features = {
        'mfcc': [],           # 梅尔频率倒谱系数
        'spectral_centroid': 0.0,  # 频谱质心
        'spectral_rolloff': 0.0,   # 频谱滚降
        'zero_crossing_rate': 0.0, # 过零率
        'tempo': 0.0,         # 节拍
        'pitch': 0.0,         # 音调
        'energy': 0.0,        # 能量
        'formants': []        # 共振峰
    }

    try:
        # 提取MFCC特征
        features['mfcc'] = librosa.feature.mfcc(y=audio, sr=self.config.get('sample_rate', 16000)).tolist()

        # 提取频谱特征
        features['spectral_centroid'] = np.mean(librosa.feature.spectral_centroid(y=audio))
        features['spectral_rolloff'] = np.mean(librosa.feature.spectral_rolloff(y=audio))
        features['zero_crossing_rate'] = np.mean(librosa.feature.zero_crossing_rate(audio))

        # 提取节拍和音调
        tempo, beats = librosa.beat.beat_track(y=audio)
        features['tempo'] = tempo

        pitches, magnitudes = librosa.piptrack(y=audio)
        features['pitch'] = np.mean(pitches[pitches > 0])

        # 计算能量
        features['energy'] = np.sum(audio ** 2) / len(audio)

        # 提取共振峰
        features['formants'] = self._extract_formants(audio)

    except Exception as e:
        self.logger.error(f"声学特征提取失败: {e}")

    return features
```

#### 5.1.6 音频质量评估

```python
def _assess_audio_quality(self, audio: np.ndarray) -> Dict[str, float]:
    """
    评估音频质量

    Args:
        audio: 音频数据

    Returns:
        Dict[str, float]: 质量评估结果
    """
    quality_metrics = {
        'snr': 0.0,           # 信噪比
        'clarity': 0.0,       # 清晰度
        'noise_level': 0.0,   # 噪声水平
        'distortion': 0.0,    # 失真度
        'overall_quality': 0.0 # 总体质量
    }

    try:
        # 计算信噪比
        signal_power = np.mean(audio ** 2)
        noise_power = self._estimate_noise_power(audio)
        quality_metrics['snr'] = 10 * np.log10(signal_power / (noise_power + 1e-10))

        # 评估清晰度
        quality_metrics['clarity'] = self._calculate_clarity(audio)

        # 评估噪声水平
        quality_metrics['noise_level'] = self._estimate_noise_level(audio)

        # 评估失真度
        quality_metrics['distortion'] = self._calculate_distortion(audio)

        # 计算总体质量
        quality_metrics['overall_quality'] = np.mean([
            min(quality_metrics['snr'] / 30, 1.0),  # 归一化SNR
            quality_metrics['clarity'],
            1.0 - quality_metrics['noise_level'],
            1.0 - quality_metrics['distortion']
        ])

    except Exception as e:
        self.logger.error(f"音频质量评估失败: {e}")

    return quality_metrics
```

## 6. 综合分析器

### 6.1 MultiModalAnalyzer 多模态分析器

#### 6.1.1 文件概述

- **文件路径**: `analyzers/multimodal_analyzer.py`
- **主要功能**: 整合多种数据类型的分析结果，提供综合分析
- **设计模式**: 组合模式

#### 6.1.2 核心功能

```python
class MultiModalAnalyzer(BaseAnalyzer):
    """多模态综合分析器"""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.text_analyzer = TextAnalyzer(config.get('text', {}))
        self.image_analyzer = ImageAnalyzer(config.get('image', {}))
        self.voice_analyzer = VoiceAnalyzer(config.get('voice', {}))
        self.fusion_model = self._init_fusion_model()

    def analyze(self, multimodal_data: Dict[str, Any]) -> AnalysisResult:
        """
        分析多模态数据

        Args:
            multimodal_data: 多模态数据字典

        Returns:
            AnalysisResult: 综合分析结果
        """
        try:
            # 分别分析各模态数据
            individual_results = {}

            if 'text' in multimodal_data:
                individual_results['text'] = self.text_analyzer.analyze(multimodal_data['text'])

            if 'image' in multimodal_data:
                individual_results['image'] = self.image_analyzer.analyze(multimodal_data['image'])

            if 'voice' in multimodal_data:
                individual_results['voice'] = self.voice_analyzer.analyze(multimodal_data['voice'])

            # 多模态融合
            fused_features = self._fuse_features(individual_results)

            # 综合分析
            comprehensive_analysis = self._comprehensive_analysis(fused_features)

            # 生成综合报告
            analysis_report = self._generate_analysis_report(individual_results, comprehensive_analysis)

            # 合并特征
            features = {
                'individual_results': individual_results,
                'fused_features': fused_features,
                'comprehensive_analysis': comprehensive_analysis,
                'analysis_report': analysis_report
            }

            # 计算综合置信度
            confidence = self._calculate_fusion_confidence(individual_results)

            return AnalysisResult.success(
                data_type='multimodal',
                features=features,
                confidence=confidence,
                metadata={'modalities': list(multimodal_data.keys())}
            )

        except Exception as e:
            self.logger.error(f"多模态分析失败: {e}")
            return AnalysisResult.error(str(e))
```

#### 6.1.3 特征融合

```python
def _fuse_features(self, individual_results: Dict[str, AnalysisResult]) -> Dict[str, Any]:
    """
    融合多模态特征

    Args:
        individual_results: 各模态分析结果

    Returns:
        Dict[str, Any]: 融合后的特征
    """
    fused_features = {
        'text_features': {},
        'image_features': {},
        'voice_features': {},
        'cross_modal_features': {},
        'fusion_weights': {}
    }

    # 提取各模态特征
    if 'text' in individual_results and individual_results['text'].success:
        fused_features['text_features'] = individual_results['text'].features

    if 'image' in individual_results and individual_results['image'].success:
        fused_features['image_features'] = individual_results['image'].features

    if 'voice' in individual_results and individual_results['voice'].success:
        fused_features['voice_features'] = individual_results['voice'].features

    # 跨模态特征提取
    fused_features['cross_modal_features'] = self._extract_cross_modal_features(individual_results)

    # 计算融合权重
    fused_features['fusion_weights'] = self._calculate_fusion_weights(individual_results)

    return fused_features
```

#### 6.1.4 综合分析

```python
def _comprehensive_analysis(self, fused_features: Dict[str, Any]) -> Dict[str, Any]:
    """
    综合分析

    Args:
        fused_features: 融合后的特征

    Returns:
        Dict[str, Any]: 综合分析结果
    """
    analysis = {
        'medical_consistency': 0.0,    # 医学一致性
        'diagnostic_support': 0.0,     # 诊断支持度
        'risk_assessment': 0.0,        # 风险评估
        'recommendations': [],         # 建议
        'confidence_score': 0.0        # 置信度评分
    }

    # 医学一致性分析
    analysis['medical_consistency'] = self._analyze_medical_consistency(fused_features)

    # 诊断支持度分析
    analysis['diagnostic_support'] = self._analyze_diagnostic_support(fused_features)

    # 风险评估
    analysis['risk_assessment'] = self._assess_risk(fused_features)

    # 生成建议
    analysis['recommendations'] = self._generate_recommendations(fused_features)

    # 计算综合置信度
    analysis['confidence_score'] = np.mean([
        analysis['medical_consistency'],
        analysis['diagnostic_support'],
        1.0 - analysis['risk_assessment']
    ])

    return analysis
```

```python
def validate_analysis_result(result: AnalysisResult) -> bool:
    """
    验证分析结果

    Args:
        result: 分析结果

    Returns:
        bool: 验证是否通过
    """
    # 检查基本字段
    if not hasattr(result, 'success') or not hasattr(result, 'data_type'):
        return False

    # 检查成功结果
    if result.success:
        if not hasattr(result, 'features') or not hasattr(result, 'confidence'):
            return False

        # 检查置信度范围
        if not (0.0 <= result.confidence <= 1.0):
            return False

    # 检查失败结果
    else:
        if not hasattr(result, 'error_message'):
            return False

    return True
```

## 8. 性能优化

### 8.1 并行处理

#### 8.1.1 多线程分析

```python
def parallel_analyze(self, data_list: List[Any], max_workers: int = 4) -> List[AnalysisResult]:
    """
    并行分析数据

    Args:
        data_list: 数据列表
        max_workers: 最大工作线程数

    Returns:
        List[AnalysisResult]: 分析结果列表
    """
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = [executor.submit(self.analyze, data) for data in data_list]
        results = [future.result() for future in as_completed(futures)]

    return results
```

#### 8.1.2 批处理优化

```python
def batch_analyze_optimized(self, data_list: List[Any], batch_size: int = 32) -> List[AnalysisResult]:
    """
    优化的批处理分析

    Args:
        data_list: 数据列表
        batch_size: 批处理大小

    Returns:
        List[AnalysisResult]: 分析结果列表
    """
    results = []

    for i in range(0, len(data_list), batch_size):
        batch = data_list[i:i + batch_size]
        batch_results = self._process_batch(batch)
        results.extend(batch_results)

    return results
```

### 8.2 内存优化

#### 8.2.1 内存管理

```python
class MemoryOptimizedAnalyzer(BaseAnalyzer):
    """内存优化的分析器"""

    def __init__(self, config: Dict[str, Any]):
        super().__init__(config)
        self.memory_limit = config.get('memory_limit', 4 * 1024 * 1024 * 1024)  # 4GB
        self.current_memory = 0

    def analyze(self, data: Any) -> AnalysisResult:
        """内存优化的分析方法"""
        # 检查内存使用
        if self.current_memory > self.memory_limit:
            self._cleanup_memory()

        # 执行分析
        result = super().analyze(data)

        # 更新内存使用
        self.current_memory += self._estimate_memory_usage(data)

        return result

    def _cleanup_memory(self):
        """清理内存"""
        import gc
        gc.collect()
        self.current_memory = 0
```
