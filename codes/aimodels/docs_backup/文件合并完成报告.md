### 编程接口

```python
from model_config_reader import ModelConfigReader

# 创建管理器实例
manager = ModelConfigReader()

# 检查模型状态
if manager.check_model_exists("BiomedCLIP-PubMedBERT_256-vit_base_patch16_224"):
    print("模型存在")

# 下载模型
manager.download_model("Qwen2-0.5B-Medical-MLX")

# 列出所有模型
models = manager.list_models()
for model_type, model_list in models.items():
    print(f"{model_type}: {len(model_list)} 个模型")
```
