```json
      "model_name": "BiomedCLIP-PubMedBERT_256-vit_base_patch16_224",
      "model_path": "/path/to/BiomedCLIP-PubMedBERT_256-vit_base_patch16_224",
      "huggingface_id": "OFA-Sys/BiomedCLIP-PubMedBERT_256-vit_base_patch16_224",
      "vector_dim": 512
    },
    "summarization": {
      "model_name": "Falconsai_text_summarization",
      "model_path": "/path/to/Falconsai_text_summarization",
      "huggingface_id": "Falconsai/text_summarization"
    },
    "llm": {
      "model_name": "Qwen2-0.5B-Medical-MLX",
      "model_path": "/path/to/Qwen2-0.5B-Medical-MLX",
      "huggingface_id": "Qwen/Qwen2-0.5B-Medical-MLX"
    }
  }
}
```
